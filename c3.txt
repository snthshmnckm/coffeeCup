6. Multiply of two matrix


#include <stdio.h>

int main() {
    int m, n, p, q;

    // Get the size of the first matrix
    printf("Enter the number of rows and columns of the first matrix: ");
    scanf("%d %d", &m, &n);

    // Get the size of the second matrix
    printf("Enter the number of rows and columns of the second matrix: ");
    scanf("%d %d", &p, &q);

    // Ensure the matrices can be multiplied (columns in first = rows in second)
    if (n != p) {
        printf("Matrices cannot be multiplied\n");
        return 0;
    }

    int matrix1[m][n], matrix2[p][q], product[m][q];

    // Initialize the product matrix to zero
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < q; j++) {
            product[i][j] = 0;
        }
    }

    // Input elements for the first matrix
    printf("Enter elements of the first matrix:\n");
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            scanf("%d", &matrix1[i][j]);
        }
    }

    // Input elements for the second matrix
    printf("Enter elements of the second matrix:\n");
    for (int i = 0; i < p; i++) {
        for (int j = 0; j < q; j++) {
            scanf("%d", &matrix2[i][j]);
        }
    }

    // Multiply the matrices
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < q; j++) {
            for (int k = 0; k < n; k++) {
                product[i][j] += matrix1[i][k] * matrix2[k][j];
            }
        }
    }

    // Print the product matrix
    printf("Product of the matrices:\n");
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < q; j++) {
            printf("%d ", product[i][j]);
        }
        printf("\n");
    }

    return 0;
}

/**
 * C program to perform scalar matrix multiplication
 */
 
#include <stdio.h>

#define SIZE 3 // Maximum size of the array

int main()
{
    int A[SIZE][SIZE]; 
    int num, row, col;

    /* Input elements in matrix from user */
    printf("Enter elements in matrix of size %dx%d: \n", SIZE, SIZE);
    for(row=0; row<SIZE; row++)
    {
        for(col=0; col<SIZE; col++)
        {
            scanf("%d", &A[row][col]);
        }
    }

    /* Input multiplier from user */
    printf("Enter any number to multiply with matrix A: ");
    scanf("%d", &num);

    /* Perform scalar multiplication of matrix */
    for(row=0; row<SIZE; row++)
    {
        for(col=0; col<SIZE; col++)
        {
            /* (cAij) = c . Aij */
            A[row][col] = num * A[row][col];
        }
    }

    /* Print result of scalar multiplication of matrix */
    printf("\nResultant matrix c.A = \n");
    for(row=0; row<SIZE; row++)
    {
        for(col=0; col<SIZE; col++)
        {
            printf("%d ", A[row][col]);
        }
        printf("\n");
    }

    return 0;
}

    // Get the size of the array

    scanf("%d", &n);

    int arr[n];

    // Input elements for the array

    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    // Get the number to remove
    printf("Enter the number to remove: ");
    scanf("%d", &num_to_remove);
	
     int found=0;
    for (int i = 0; i < n; i++) {
        if (arr[i] == num_to_remove) {
            found = 1;
            // Shift elements to the left to remove the element
            for (int j = i; j < n - 1; j++) {
                arr[j] = arr[j + 1];
            }
            n--; // Decrease the size of the array
            break; // Break to ensure only the first occurrence is removed
        }
    }

    if (!found) {
        printf("Number not found in the array.\n");
    } else {
        // Print the updated array
        printf("Updated array after removing the first occurrence of %d:\n", num_to_remove);
        for (int i = 0; i < n; i++) {
            printf("%d ", arr[i]);
        }
        printf("\n");
    }

12.deletion of duplicate elements in array


#include <stdio.h>

int main() {
    int n;

    // Get the size of the array
    printf("Enter the number of elements in the array: ");
    scanf("%d", &n);

    int arr[n];

    // Input elements for the array
    printf("Enter the elements of the array:\n");
    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    // Remove duplicate elements
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] == arr[j]) {
                // Shift elements to the left to remove the duplicate
                for (int k = j; k < n - 1; k++) {
                    arr[k] = arr[k + 1];
                }
                n--; // Decrease the size of the array
                j--; // Adjust the index to check the next element
            }
        }
    }

    // Print the updated array
    printf("Updated array without duplicates:\n");
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");

    return 0;
}
